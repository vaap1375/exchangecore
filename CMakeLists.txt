cmake_minimum_required(VERSION 3.20)
project(ExchangeCore VERSION 1.0.0 LANGUAGES CXX)

include(CTest)
include(GNUInstallDirs)
option(EXCHANGECORE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(EXCHANGECORE_ENABLE_INVARIANTS "Check all book invariants after each event (expensive)" OFF)
option(EXCHANGECORE_BUILD_BENCHMARKS "Build standalone allocation-instrumented benchmarks" ON)

add_library(exchangecore_options INTERFACE)
target_compile_features(exchangecore_options INTERFACE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(exchangecore_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(EXCHANGECORE_SANITIZERS)
        target_compile_options(exchangecore_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(exchangecore_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
elseif(MSVC)
    target_compile_options(exchangecore_options INTERFACE /W4 /permissive-)
    if(EXCHANGECORE_SANITIZERS)
        message(FATAL_ERROR "Use Clang or GCC for the sanitizer configuration")
    endif()
endif()
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(exchangecore_lib src/order_book.cpp src/matching_engine.cpp src/csv.cpp src/price.cpp)
add_library(ExchangeCore::exchangecore ALIAS exchangecore_lib)
set_target_properties(exchangecore_lib PROPERTIES OUTPUT_NAME exchangecore EXPORT_NAME exchangecore)
target_include_directories(exchangecore_lib PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(exchangecore_lib PUBLIC exchangecore_options)
if(EXCHANGECORE_ENABLE_INVARIANTS)
    target_compile_definitions(exchangecore_lib PRIVATE EXCHANGECORE_ENABLE_INVARIANTS=1)
endif()

add_library(exchangecore_benchmark_support STATIC benchmarks/benchmark.cpp)
target_link_libraries(exchangecore_benchmark_support PUBLIC exchangecore_lib)
add_executable(exchangecore src/main.cpp)
target_link_libraries(exchangecore PRIVATE exchangecore_benchmark_support)
if(EXCHANGECORE_BUILD_BENCHMARKS)
    add_executable(exchangecore_bench benchmarks/main.cpp benchmarks/allocation_counter.cpp)
    target_link_libraries(exchangecore_bench PRIVATE exchangecore_benchmark_support)
endif()

if(BUILD_TESTING)
    add_executable(exchangecore_tests tests/test_main.cpp)
    target_link_libraries(exchangecore_tests PRIVATE exchangecore_lib)
    add_test(NAME core_and_reference COMMAND exchangecore_tests)
    set_tests_properties(core_and_reference PROPERTIES TIMEOUT 180)
    add_executable(exchangecore_exception_tests tests/exception_test.cpp)
    target_link_libraries(exchangecore_exception_tests PRIVATE exchangecore_lib)
    add_test(NAME allocation_failure_rollback COMMAND exchangecore_exception_tests)
    add_executable(exchangecore_benchmark_tests tests/benchmark_test.cpp)
    target_link_libraries(exchangecore_benchmark_tests PRIVATE exchangecore_benchmark_support)
    add_test(NAME benchmark_generators COMMAND exchangecore_benchmark_tests)
    add_test(NAME cli_demo COMMAND exchangecore demo)
    add_test(NAME cli_help COMMAND exchangecore --help)
    add_test(NAME cli_validate COMMAND exchangecore validate ${CMAKE_CURRENT_SOURCE_DIR}/examples/orders.csv)
    add_test(NAME cli_replay COMMAND exchangecore replay ${CMAKE_CURRENT_SOURCE_DIR}/examples/orders.csv)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME cli_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:exchangecore>)
    endif()
endif()

# A dependency-free installed library consumable with find_package(ExchangeCore).
install(TARGETS exchangecore_lib exchangecore_options EXPORT ExchangeCoreTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/exchangecore DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
    PATTERN benchmark.hpp EXCLUDE)
install(EXPORT ExchangeCoreTargets NAMESPACE ExchangeCore::
    FILE ExchangeCoreConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ExchangeCore)
